  fn Determinant(sf) -> (float64, error) {

      rt (0, fmt.Errorf("matrix must be square"))
    }
mt det = go_var("float64")
    if sf.Rows == 1 {
      rt (sf.Data[0][0], nil)
    } else if sf.Rows == 2 {
      mt a = sf.Data[0][0]
      mt b = sf.Data[0][1]
      mt c = sf.Data[1][0]
      mt d = sf.Data[1][1]
      rt (a * d - b * c, nil)
    } else {
mt i = 0
      wl i < sf.Rows {
        mt submatrix = NewMatrix(sf.Rows - 1, sf.Cols - 1)
mt j = 1
        wl j < sf.Rows {
mt k = 0
          wl k < sf.Cols {
            if k != i {
              submatrix.Set(j - 1, k < i ? k : k - 1, sf.Data[j][k])
            }
            k += 1
          }
          j += 1
        }
        mt sign = math.Pow(-1, float64(i))
mt (subdet, err) = submatrix.Determinant()
        if err != nil {
          rt (0, err)
        }
        det += sf.Data[0][i] * sign * subdet
        i += 1
      }
      rt (det, nil)
    }
  }
}
fn LinearRegression(xs: List<float64>, ys: List<float64>) -> (float64, float64, error) {
  if ln(xs) != ln(ys) or ln(xs) == 0 {
    rt (0, 0, fmt.Errorf("input lengths must match and be non-zero"))
  }
mt sumX = go_var("float64")
mt sumY = go_var("float64")
mt sumXY = go_var("float64")
mt sumXX = go_var("float64")
mt n = ln(xs)
mt i = 0
  wl i < n {
    mt x = xs[i]
    mt y = ys[i]
    sumX += x
    sumY += y
    sumXY += x * y
    sumXX += x * x
    i += 1
  }
mt slope = (n * sumXY - sumX * sumY) / (n * sumXX - sumX * sumX)
mt intercept = (sumY - slope * sumX) / n
  rt (slope, intercept, nil)
}
