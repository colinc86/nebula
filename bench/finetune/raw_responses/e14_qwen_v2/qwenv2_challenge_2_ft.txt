im Graph {
  fn ShortestPath(sf, from: int, to: int) -> (List<int>, float64, error) {
    mt visited = make("map[int]bool")
    mt distances = make("map[int]float64")
    mt previousNodes = make("map[int]int")
    fo _, n in sf.nodes {
      distances[n.ID] = math.MaxFloat64
    }
    distances[from] = 0.0
    mt priorityQueue = new(PriorityQueue)
    priorityQueue.Push(from, 0.0)
    wh not priorityQueue.IsEmpty() {
      mt (nodeID, distance) = priorityQueue.Pop()
      if nodeID == to {
        rt sf.reconstructPath(previousNodes, from, to), distance, nil
      }
      visited[nodeID] = tt
      fo _, e in sf.edges {
        if e.From == nodeID and not visited[e.To] {
          mt newDistance = distance + e.Cost
          if newDistance < distances[e.To] {
            distances[e.To] = newDistance
            previousNodes[e.To] = nodeID
            priorityQueue.Push(e.To, newDistance)
          }
        }
      }
    }
    rt nil, 0.0, ErrNoPath(from, to)
  }

  fn reconstructPath(sf, previous: Map<int, int>, start: int, end: int) -> List<int> {
    mt path = go_var("[]int")
    for current := end; current != start; current = previous[current] {
      path = psh(path, current)
    }
    path = psh(path, start)
    rt reverse(path)
  }

        components = psh(components, sort(component))
      }
    }
    rt components
  }
