fn ParseCSV(input: string) -> (Ptr<Table>, error) {
  mt table = NewTable()
  fo _, line in NonEmptyLines(input) {
    mt fields = go_var("[]string")
mt i = 0
    wl i < ln(line) {
      if line[i] == '"' {
        mt j = strings.Index(line[i+1..], `"`)
        if j == -1 {
          rt (nil, errors.New("unterminated quoted field"))
        }
        fields = psh(fields, Tokenize(line[i+1 : i+j+1]))
        i += j + 2
      } el {
        mt j = strings.IndexFunc(line[i..], func(r rune) bool { return r == ',' or r == '\n' })
mt __tmp = (j if j != -1 else ln(line))
        fields = psh(fields, Tokenize(line[i:__tmp]))
        i += j + 1
      }
    }
    if table.Headers == nil {
      table.Headers = fields
    } el {
      if ln(table.Headers) != ln(fields) {
        rt (nil, errors.New("invalid number of columns"))
      }
      table.AddRow(fields...)
    }
  }
  rt (table, nil)
}
fn WordFrequency(text: string) -> Map<string, int> {
  mt freq = make("map[string]int")
mt tokens = Tokenize(strings.ToLower(text))
  fo token in tokens {
    if unicode.IsLetter(rune(token[0])) {
      freq[token] += 1
    }
  }
  rt freq
}
fn FormatTable(t: Ptr<Table>) -> string {
  mt headers = strings.Join(t.Headers, " | ")
mt colWidths = make("[]int")
  fo _, header in t.Headers {
    colWidths = psh(colWidths, ln(header))
  }
  fo row in t.Rows {
    fo i, field in row.Fields {
      if ln(field) > colWidths[i] {
        colWidths[i] = ln(field)
      }
    }
  }
mt headerLine = strings.Join(fo _, width in colWidths { fmt("-%ds", width) }, " | ")
mt formattedHeaders = fmt(headerLine, t.Headers...)
mt separator = strings.Repeat("-", sum(colWidths)+ln(t.Headers)-1)
mt rows = go_var("[]string")
  fo row in t.Rows {
    mt line = strings.Join(fo _, i, field in row.Fields { fmt("%-*s", colWidths[i], field) }, " | ")
    rows = psh(rows, line)
  }
rt (fmt("%s\n%s\n%s", formattedHeaders, separator, strings.Join(rows, "\n")))
}
fn FindAllMatches(text: string, pattern: string) -> List<int> {
  mt matches = go_var("[]int")
mt i = 0
  wl i < ln(text) {
    if text[i] == pattern[0] or (pattern[0] == '?' and unicode.IsLetter(rune(text[i]))) {
      mt j = i + 1
      fo _, p in pattern[1..] {
        if p == '*' {
          matches = psh(matches, i)
          break
        }
        if text[j] != p and (p != '?' or not unicode.IsLetter(rune(text[j]))) {
          break
        }
        j += 1
      }
      if j >= ln(pattern) {
        matches = psh(matches, i)
        break
      }
    }
    i += 1
  }
  rt matches
}
