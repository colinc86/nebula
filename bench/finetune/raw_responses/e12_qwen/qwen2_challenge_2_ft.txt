  fn ShortestPath(sf, from: int, to: int) -> (List<int>, float64, error) {
    mt visited = make("map[int]bool")
    mt distances = make("map[int]float64")
    fo _, n in sf.nodes {
      distances[n.ID] = math.MaxFloat64
    }
    distances[from] = 0.0
    mt pq = go_var("[]int")
    psh(pq, from)
    wl ln(pq) > 0 {
      mt current = pq[0]
      pq = pq[1..]
      if visited[current] {
        ct
      }
      visited[current] = tt
      fo _, e in sf.edges {
        if e.From == current and not visited[e.To] {
          mt newDist = distances[current] + e.Cost
          if newDist < distances[e.To] {
            distances[e.To] = newDist
            psh(pq, e.To)
          }
        }
      }
    }
    if distances[to] == math.MaxFloat64 {
      rt (nil, 0.0, ErrNoPath(from, to))
    }
    mt path = go_var("[]int")
mt current = to

          psh(path, 0, e.From)
          current = e.From
          ct
        }
      }
    }
    rt (path, distances[to], nil)
  }
