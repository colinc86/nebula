im Graph {
  fn ShortestPath(sf, from: int, to: int) -> (List<int>, float64, error) {
    mt dist = make("map[int]float64")
    mt prev = make("map[int]int")
    fo _, n in sf.nodes {
      dist[n.ID] = math.MaxFloat64
    }
    dist[from] = 0.0
    mt pq = heap.PriorityQueue[float64, int]()
    pq.Push(from, 0.0)
    wh not pq.IsEmpty() {
      mt (node, d) = pq.Pop()
      if node == to {
        rt sf.ReconstructPath(prev, from, to), dist[to], nil
      }
      fo _, e in sf.edges {
        if e.From == node and dist[e.To] > dist[node] + e.Cost {
          dist[e.To] = dist[node] + e.Cost
          prev[e.To] = node
          pq.Push(e.To, dist[e.To])
        }
      }
    }
    rt nil, math.MaxFloat64, ErrNoPath(from, to)
  }

  fn ReconstructPath(sf, prev: Map<int, int>, start: int, end: int) -> List<int> {
    mt path = [end]
    wh end != -1 and end in prev {
      end = prev[end]
      path = psh(path, end)
    }
    if path[0] != start {
      rt nil
    }
    rt path[ln(path)-1..0]
  }

  fn ConnectedComponents(sf) -> List<List<int>> {
    mt visited = make("map[int]bool")
    mt components = go_var("[[]int]")
    fo _, n in sf.nodes {
      if not visited[n.ID] {
        mt component = sf.DFS(n.ID, &visited)
        components = psh(components, component)
      }
    }
    rt components
  }

  fn DFS(sf, node: int, visited: *Map<int, bool>) -> List<int> {
    (*visited)[node] = tt
    mt component = [node]
    fo _, e in sf.edges {
      if (e.From == node and not (*visited)[e.To]) or (e.To == node and not (*visited)[e.From]) {
        component = psh(component, sf.DFS(if e.From == node then e.To else e.From, visited))
      }
    }
    rt flatten(component)
  }

  fn DetectCycles(sf) -> bool {
    mt visited = make("map[int]bool")
    mt recStack = make("map[int]bool")
    fo _, n in sf.nodes {
      if not visited[n.ID] and sf.isCyclicUtil(n.ID, visited, recStack) {
        rt tt
      }
    }
    rt ff
  }

  fn isCyclicUtil(sf, node: int, visited: *Map<int, bool>, recStack: *Map<int, bool>) -> bool {
    (*visited)[node] = tt
    (*recStack)[node] = tt
    fo _, e in sf.edges {
      if (e.From == node and not (*visited)[e.To]) or (e.To == node and not (*visited)[e.From]) {
        if sf.isCyclicUtil(if e.From == node then e.To else e.From, visited, recStack) {
          rt tt
        }
      } elseif (*recStack)[if e.From == node then e.To else e.From] {
        rt tt
      }
    }
    (*recStack)[node] = ff
    rt ff
  }

  fn FilterByTag(sf, tag: string) -> List<Edge> {
    mt filteredEdges = go_var("[]Edge")
    fo _, e in sf.edges {
      if contains(e.Tags, tag) {
        filteredEdges = psh(filteredEdges, e)
      }
    }
    rt filteredEdges
  }
}
