fn ParseCSV(input: string) -> (Ptr<Table>, error) {
  mt lines = NonEmptyLines(input)
  if ln(lines) == 0 {
    rt (nil, errors.New("empty CSV"))
  }
  mt headers = Tokenize(lines[0])
  mt table = NewTable(headers...)
  fo _, line in lines[1:] {
    mt fields = go_var("[string]")
    mt err = csv.ParseFields(line, &fields)
    if err != nil {
      rt (nil, errors.New("invalid CSV"))
    }
    if ln(fields) != ln(headers) {
      rt (nil, errors.New("mismatched columns"))
    }
    table.AddRow(fields...)
  }
  rt (table, nil)
}
fn WordFrequency(text: string) -> Map<string, int> {
  mt freq = make("map[string]int")
  fo _, word in Tokenize(strings.ToLower(text)) {
    freq[word] += 1
  }
  rt freq
}
fn FormatTable(t: Ptr<Table>) -> string {
  if t == nil or ln(t.Headers) == 0 {
    return ""
  }
  mt colWidths = make("[]int, len(t.Headers)")
mt (i, j) = (0, len(colWidths))
  wl i < j {
    colWidths[i] = 4
    i += 1
  }
  fo _, row in t.Rows {
    fo i, field in row.Fields {
      if ln(field) > colWidths[i] {
        colWidths[i] = ln(field)
      }
    }
  }
mt (i, j) = (0, len(t.Headers))
  wl i < j {
    mt header = fmt.Sprintf("%-*s", colWidths[i], t.Headers[i])
    if i == 0 {
      rt header
    }
    rt header + " | "
  }
}
fn FindAllMatches(text: string, pattern: string) -> List<int> {
  mt matches = go_var("[]int")
mt (i, j) = (0, len(text))
  wl i < j {
    if strings.HasPrefix(pattern, "*") {
      mt index = strings.Index(text[i:], pattern[1:])
      if index == -1 {
        rt matches
      }
      matches = psh(matches, i + index)
      i += ln(pattern) - 2
    } el if strings.HasPrefix(pattern, "?") {
      if len(text) > i and (pattern[1] == text[i] or pattern[1] == '?') {
        matches = psh(matches, i)
        i += 1
      }
    } el {
      mt index = strings.Index(text[i:], pattern)
      if index == -1 {
        rt matches
      }
      matches = psh(matches, i + index)
      i += ln(pattern) - 1
    }
    pattern = pattern[ln(pattern)-1:]
  }
  rt matches
}
