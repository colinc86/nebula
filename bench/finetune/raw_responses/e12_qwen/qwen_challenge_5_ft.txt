im Matrix {
  fn Multiply(sf, other: Ptr<Matrix>) -> (Ptr<Matrix>, error) {
    if sf.Cols != other.Rows {
      rt (nil, fmt.Errorf("dimension mismatch: %dx%d vs %dx%d", sf.Rows, sf.Cols, other.Rows, other.Cols))
    }
    mt result = NewMatrix(sf.Rows, other.Cols)
mt i = 0
    wl i < sf.Rows {
mt j = 0
      wl j < other.Cols {
        mt sum = go_var("float64")
mt k = 0
        wl k < sf.Cols {
          sum += sf.Data[i][k] * other.Data[k][j]
          k += 1
        }
        result.Data[i][j] = sum
        j += 1
      }
      i += 1
    }
    rt (result, nil)
  }
}

im Matrix {
  fn Determinant(sf) -> (float64, error) {
    if sf.Rows != sf.Cols {
      rt (0.0, fmt.Errorf("matrix must be square"))
    }
mt n = sf.Rows
    mt det = go_var("float64")
    if n == 1 {
      return sf.Data[0][0], nil
    } else if n == 2 {
      det = sf.Data[0][0] * sf.Data[1][1] - sf.Data[0][1] * sf.Data[1][0]
      rt (det, nil)
    }
mt cofactors = make("[][]float64, n")
mt i = 0
    wl i < n {
mt j = 0
      wl j < n {
        mt submatrix = NewMatrix(n - 1, n - 1)
mt k = 0
        wl k < n {
          if k == i {
            continue
          }
mt l = 0
          wl l < n {
            if l == j {
              continue
            }
            submatrix.Data[k < i ? k : k - 1][l < j ? l : l - 1] = sf.Data[k][l]
            l += 1
          }
          k += 1
        }
mt (_, err) = submatrix.Determinant()
        if err != nil {
          rt (0.0, err)
        }
        cofactors[i][j] = math.Pow(-1, float64(i+j)) * submatrix.Data[0][0]
        j += 1
      }
      i += 1
    }
mt sum = go_var("float64")
mt j = 0
    wl j < n {
      sum += sf.Data[0][j] * cofactors[0][j]
      j += 1
    }
    rt (sum, nil)
  }
}
fn LinearRegression(xs: List<float64>, ys: List<float64>) -> (float64, float64, error) {
  if ln(xs) != ln(ys) or ln(xs) == 0 {
    rt (0.0, 0.0, fmt.Errorf("input lengths must match and be non-zero"))
  }
mt n = ln(xs)
  mt sumX = go_var("float64")
  mt sumY = go_var("float64")
  fo _, x in xs {
    sumX += x
  }
fo _, y in ys {
    sumY += y
  }
  mt meanX = sumX / float64(n)
  mt meanY = sumY / float64(n)
mt sumXY = go_var("float64")
  fo i in xs {
    sumXY += (xs[i] - meanX) * (ys[i] - meanY)
  }
mt sumXX = go_var("float64")
  fo _, x in xs {
    sumXX += math.Pow(x - meanX, 2.0)
  }
  mt slope = sumXY / sumXX
  mt intercept = meanY - slope * meanX
  rt (slope, intercept, nil)
}
