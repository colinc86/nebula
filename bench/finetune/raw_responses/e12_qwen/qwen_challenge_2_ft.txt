im Graph {
  fn ShortestPath(sf, from: int, to: int) -> (List<int>, float64, error) {
    mt dist = make("map[int]float64")
    fo _, n in sf.nodes {
      dist[n.ID] = math.MaxFloat64
    }
    dist[from] = 0.0
    mt pq = pqueue.New()
    pq.Push(Pair { Value: from, Priority: 0.0 })
    wl pq.Len() > 0 {
      mt node = pq.Pop().(Pair).Value.(int)
      if node == to {
        rt (nil, dist[to], nil)
      }
      fo _, e in sf.edges {
        if e.From != node {
          ct
        }
        mt alt = dist[node] + e.Cost
        if alt < dist[e.To] {
          dist[e.To] = alt
          pq.Push(Pair { Value: e.To, Priority: alt })
        }
      }
    }
    rt (nil, 0.0, ErrNoPath(from, to))
  }
}
